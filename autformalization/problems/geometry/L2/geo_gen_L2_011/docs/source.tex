\begin{theorem}[isInteriorPoint_of_bijective_mfderiv]
Suppose $M$ is a smooth manifold (without boundary), $N$ is a smooth manifold with boundary, and $F: M \to N$ is smooth. Show that if $p \in M$ is a point such that $dF_p$ is nonsingular, then $F(p) \in \operatorname{Int} N$.
\end{theorem}
