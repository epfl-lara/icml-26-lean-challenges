\begin{theorem}[napoleons_theorem_inner]\label{thm:napoleon_inner}
Let $A,B,C$ be non-collinear points in the plane.
On each side of $\triangle ABC$, construct an equilateral triangle \emph{internally} (all three on the same ``inner'' side).
Let $X,Y,Z$ be the centroids of the equilateral triangles on $AB,BC,CA$, respectively.
Then $\triangle XYZ$ is equilateral.
\end{theorem}
