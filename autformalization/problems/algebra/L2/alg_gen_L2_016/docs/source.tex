\begin{theorem}[span_pow_card_mul_le_span_image_pow]
Let $s=\{g_1,\dots,g_r\}$ be a finite subset of $k[x_1,\dots,x_n]$.
Set $J := \langle s\rangle$ and let $M\ge 0$. Then
\[
  J^{\,rM}
  \;\subseteq\;
  \left\langle\, g^M \mid g\in s \,\right\rangle.
\]
\end{theorem}
