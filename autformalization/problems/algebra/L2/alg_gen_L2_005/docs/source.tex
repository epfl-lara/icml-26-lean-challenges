\begin{definition}[S]
Let $k$ be an arbitrary field and let $S$ be the subset of all polynomials in $k[x_1,\dots,x_n]$ that have no zeros in $k^n$. If $I$ is any ideal in $k[x_1,\dots,x_n]$ such that $I \cap S=\varnothing$, show that $\mathbf V(I)\neq\varnothing$.
\end{definition}
